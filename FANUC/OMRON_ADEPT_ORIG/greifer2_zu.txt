.PROGRAM greifer2_zu()
        TYPE "Close gripper 2"
        SIGNAL a_greifer2_zu, -a_greifer2_auf
        TIMER 1 = 0

        WHILE NOT SIG(e_greifer_2_zu) DO
            WAIT
            WAIT
            IF TIMER(1) > 2 THEN
                TYPE "Failed to close gripper 2"
                DO
                    WAIT
                    WAIT
                    DISABLE POWER
                UNTIL NOT SWITCH(POWER)
            END         ;if
        END ;while

        TIMER 1 = 0
        WAIT TIMER(1) > 0.1


.END
